1. Installer

uv add xui
uv pip install -e ../microframe   # le vrai moteur (pas le paquet PyPI "microframe", sans rapport)

2. Câbler l'app hôte (main.py)

from xui.csrf import CSRFMiddleware
from xui.mount import mount_builtin_assets

app.add_middleware(CSRFMiddleware, get_token=lambda: engine.csrf_token,
                    protected_paths=["/plugins/mon_plugin"])

# dans lifespan(), après await xcore.boot(app) :
mount_builtin_assets(app)  # CSS/JS des  livrés avec xui

3. Écrire un plugin avec une page

from xui.urls import path, mount_xui_pages

def _home_view(ctx):
    ctx.require_role("app.view")
    return {"title": "Accueil"}

urlpatterns = [path("/", _home_view, template="mon_plugin/home.html", name="mon_plugin.home")]

class Plugin(TrustedBase):
    def get_router(self):
        router = APIRouter()
        engine = self.get_service("ext.template_engine").engine
        mount_xui_pages(router, self.ctx, engine, urlpatterns, login_path="/plugins/demo_auth/login")
        return router
